// === BASIC ARITHMETIC FUNCTIONS ===
I64 Add(I64 a, I64 b) { return a + b; }
I64 Sub(I64 a, I64 b) { return a - b; }
I64 Mul(I64 a, I64 b) { return a * b; }
I64 Div(I64 a, I64 b) { return a / b; }
I64 Mod(I64 a, I64 b) { return a % b; }

// === LOGICAL FUNCTIONS ===
I64 And(I64 a, I64 b) { return a && b; }
I64 Or(I64 a, I64 b) { return a || b; }
I64 Not(I64 a) { return !a; }

// === COMPARISON FUNCTIONS ===
I64 Eq(I64 a, I64 b) { return a == b; }
I64 Ne(I64 a, I64 b) { return a != b; }
I64 Lt(I64 a, I64 b) { return a < b; }
I64 Le(I64 a, I64 b) { return a <= b; }
I64 Gt(I64 a, I64 b) { return a > b; }
I64 Ge(I64 a, I64 b) { return a >= b; }

// === BITWISE FUNCTIONS ===
I64 BitAnd(I64 a, I64 b) { return a & b; }
I64 BitOr(I64 a, I64 b) { return a | b; }
I64 BitXor(I64 a, I64 b) { return a ^ b; }
I64 Shl(I64 a, I64 b) { return a << b; }
I64 Shr(I64 a, I64 b) { return a >> b; }

// === CONTROL FLOW FUNCTIONS ===
I64 Max(I64 a, I64 b) { return a > b ? a : b; }
I64 Min(I64 a, I64 b) { return a < b ? a : b; }
I64 Abs(I64 x) { if (x < 0) return -x; else return x; }
I64 IsEven(I64 x) { return x % 2 == 0 ? 1 : 0; }
I64 IsOdd(I64 x) { return x % 2 == 1 ? 1 : 0; }

// === LOOP FUNCTIONS ===
I64 SumTo(I64 n) { I64 sum = 0; for (I64 i = 1; i <= n; i = i + 1) sum = sum + i; return sum; }
I64 Factorial(I64 n) { I64 result = 1; for (I64 i = 1; i <= n; i = i + 1) result = result * i; return result; }
I64 Power(I64 base, I64 exp) { I64 result = 1; for (I64 i = 0; i < exp; i = i + 1) result = result * base; return result; }

// === NESTED FUNCTION CALLS ===
I64 Complex1(I64 a, I64 b, I64 c) { return Add(Mul(a, b), Sub(c, a)); }
I64 Complex2(I64 x, I64 y) { return Max(Min(x, y), Add(x, y)); }
