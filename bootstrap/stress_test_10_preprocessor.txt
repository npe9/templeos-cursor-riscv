I64 ScopeTest(I64 base) {
  I64 result = base;
  if (base > 0) {
    I64 result = base * 2;
    result = result + 1; // inner shadowed result
  }
  {
    I64 inner = base + 5;
    result = result + inner;
  }
  for (I64 i = 0; i < 3; i = i + 1) {
    I64 temp = i * base;
    result = result + temp;
  }
  return result;
}
ScopeTest(4);

I64 NestedBlocks() {
  I64 sum = 0;
  for (I64 i = 0; i < 3; i = i + 1) {
    for (I64 j = 0; j < 3; j = j + 1) {
      sum = sum + (i * j);
    }
  }
  return sum;
}
NestedBlocks();
