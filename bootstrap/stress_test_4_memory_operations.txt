}

I64 MemoryTest() {
    I64 *arr = malloc(5 * sizeof(I64));
    if (arr == 0) return -1;
    
    for (I64 i = 0; i < 5; i = i + 1) {
        arr[i] = i * i;
    }
    
    I64 sum = 0;
    for (I64 i = 0; i < 5; i = i + 1) {
        sum = sum + arr[i];
    }
    
    free(arr);
    return sum;
}

I64 StringLength(char *str) {
    I64 len = 0;
    while (str[len] != 0) {
        len = len + 1;
